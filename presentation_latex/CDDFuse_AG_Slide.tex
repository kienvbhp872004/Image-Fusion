\section{Giới thiệu bài toán}
% =================================================================

\begin{frame}{Tổng hợp ảnh y tế đa phương thức là gì?}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \textbf{Mỗi phương thức ảnh y tế có ưu/nhược điểm riêng:}
            \begin{itemize}
                \item \textbf{MRI}: Độ phân giải cao về mô mềm, não bộ
                \item \textbf{CT}: Rõ cấu trúc xương, mô cứng
                \item \textbf{PET/SPECT}: Thông tin chức năng, trao đổi chất
            \end{itemize}
            \vspace{0.5em}
            \textbf{Medical Image Fusion (MIF)} kết hợp hai phương thức thành \alert{một ảnh duy nhất} --- giảm thời gian chẩn đoán, tăng độ chính xác.

            \vspace{0.5em}
            \textbf{Thách thức đặc thù:} Không tồn tại ảnh tổng hợp chuẩn (no ground-truth) $\Rightarrow$ hàm mất mát không giám sát.
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{figure}
                \includegraphics[width=\textwidth]{fig1_mif_examples.png}
                \caption{\small Ví dụ tổng hợp CT-MRI, PET-MRI, SPECT-MRI trên Harvard MIF.}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Tại sao chọn CDDFuse làm nền tảng?}
    \begin{columns}[T]
        \begin{column}{0.54\textwidth}
            \textbf{CDDFuse} (Zhao et al., CVPR 2023):
            \begin{itemize}
                \item Dual-branch Transformer--CNN: tách biệt đặc trưng thành nhánh \textbf{Base} (tần số thấp) và \textbf{Detail} (tần số cao)
                \item Kết hợp Restormer Encoder + Invertible Neural Network
                \item Xếp hạng \textbf{\#2/22} SOTA trên Harvard MIF (Composite Z-score = $+0.926$)
            \end{itemize}
            \vspace{0.6em}
            \textbf{Hạn chế cốt lõi:} Cả hai nhánh Base và Detail đều dùng chung \alert{phép cộng đơn giản}:
            \[ f_F = f_V + f_I \]
            Không khai thác sự \textbf{khác biệt bản chất} giữa tần số thấp và tần số cao.
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{Câu hỏi nghiên cứu}
                \textit{Nếu thay phép cộng bằng các quy tắc \textbf{chuyên biệt và khác nhau} cho mỗi nhánh --- kết quả sẽ thay đổi như thế nào?}
            \end{block}
            \vspace{0.5em}
            \begin{alertblock}{Điểm can thiệp}
                Chỉ thay thế \textbf{Fusion Layer}, giữ nguyên toàn bộ Encoder--Decoder pretrained $\Rightarrow$ so sánh công bằng, tốn ít tài nguyên.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Nguyên lý thiết kế bất đối xứng}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \textbf{Bản chất khác nhau của hai nhánh:}
            \vspace{0.4em}
            \begin{itemize}
                \item \textbf{Nhánh Base} (tần số thấp): thông tin cấu trúc toàn cục --- hai modality có \emph{tương quan cao} $\Rightarrow$ cần \textbf{trung bình hóa}
                \item \textbf{Nhánh Detail} (tần số cao): cạnh, texture đặc thù từng modality --- thông tin \emph{bổ trợ nhau} $\Rightarrow$ cần \textbf{lựa chọn cục bộ}
            \end{itemize}
            \vspace{0.5em}
            Phép cộng Sum áp dụng cùng một chiến lược cho cả hai --- \alert{không tối ưu}.
        \end{column}
        \begin{column}{0.46\textwidth}
            \begin{figure}
                \includegraphics[width=\textwidth]{fig_asymmetric_principle.png}
                \caption{\small WAvg phù hợp Base (đồng thuận); SML phù hợp Detail (bổ trợ).}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Quy tắc Base: Weighted Average Scalar (WAvg)}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \textbf{Ý tưởng:} Trung bình có trọng số với \textbf{1 scalar học được} $\theta$:
            \begin{align}
                \alpha &= \sigma(\theta), \quad \theta \in \mathbb{R} \\
                R_{\mathrm{WAvg}}(a,b) &= 2\bigl(\alpha \cdot a + (1-\alpha) \cdot b\bigr)
            \end{align}
            \vspace{0.3em}
            \textbf{Ưu điểm:}
            \begin{itemize}
                \item Chỉ \textbf{1 tham số học} --- tránh overfitting
                \item Trung bình hóa toàn cục --- phù hợp bản chất đồng thuận của Base
                \item Mô hình tự học tỉ lệ đóng góp tối ưu $\alpha$ giữa hai modality
                \item Khởi tạo $\theta=0$ ($\alpha=0.5$) --- chuyển pha ổn định
            \end{itemize}
        \end{column}
        \begin{column}{0.46\textwidth}
            \begin{figure}
                \includegraphics[width=\textwidth]{fig_rule_wavg.png}
                \caption{\small WAvg: hai Feature Map Base kết hợp bằng scalar $\alpha$ học được.}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Quy tắc Detail: Sum-Modified-Laplacian (SML)}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \textbf{Ý tưởng:} Tại mỗi vị trí, chọn modality có cạnh \textbf{sắc nét hơn}:
            \begin{align}
                \mathrm{ML}_{ij}(f) &= |2f_{ij} - f_{i-1,j} - f_{i+1,j}| \notag\\
                                    &\quad + |2f_{ij} - f_{i,j-1} - f_{i,j+1}| \\
                w_a &= \frac{\mathrm{SML}(a)}{\mathrm{SML}(a)+\mathrm{SML}(b)+\varepsilon} \\
                R_{\mathrm{SML}} &= 2\bigl(w_a \odot a + (1-w_a) \odot b\bigr)
            \end{align}
            \textbf{Ưu điểm:}
            \begin{itemize}
                \item \textbf{Không có tham số học} --- tính tổng quát cao
                \item Nhạy cảm với cạnh và tần số cao (Modified Laplacian)
                \item Lựa chọn \textbf{theo vị trí không gian} (spatially adaptive)
            \end{itemize}
        \end{column}
        \begin{column}{0.44\textwidth}
            \begin{figure}
                \includegraphics[width=\textwidth]{fig_rule_sml.png}
                \caption{\small SML: weight map $w_a$ --- xanh = ưu tiên Detail A, đỏ = ưu tiên Detail B.}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Kiến trúc CDDFuse-AG tổng thể}
    \begin{columns}[T]
        \begin{column}{0.38\textwidth}
            \textbf{Thay đổi duy nhất so với CDDFuse:}
            \begin{itemize}
                \item Nhánh Base: Sum $\rightarrow$ \textbf{WAvg}
                \item Nhánh Detail: Sum $\rightarrow$ \textbf{SML}

            \begin{block}{Hướng mở rộng}
                \textbf{Modality-aware routing}: tự động chọn fusion rule dựa trên phân phối đặc trưng đầu vào --- không cần biết trước modality.
            \end{block}
